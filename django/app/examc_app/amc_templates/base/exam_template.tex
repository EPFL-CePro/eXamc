% Pad the current copy with blank pages up to \TotalPagesPerCopy
\newcommand{\examcPadPages}{%
  \loop\ifnum\value{page}<\TotalPagesPerCopy\relax
    \clearpage\mbox{}%
  \repeat
}

% The CSV values are read with the special characters below as plain characters (see \csvreader): restore their
% LaTeX meaning before the files of the copy are read, else their % comments would be printed
\newcommand{\examcRestoreCatcodes}{%
  \catcode`\%=14 \catcode`\&=4 \catcode`\#=6 \catcode`\_=8 \catcode`\^=7 \catcode`\~=13 \catcode`\$=3 %
}

% One copy of the exam, for the current student of the CSV list
\newcommand{\examcCopy}{%
  \examcRestoreCatcodes
  \onecopy{1}{%
    \setcounter{equation}{0}%
    \input{./first_page.tex}
    %SECTIONS-INSERT%
    \examcPadPages
    \AMCassociation{\ID}%
  }%
}
